\packagename{@gorhom/bottom-sheet}~\cite{site:rn-bottomsheet}
\newline
\textit{Version:} \texttt{\^{}5.2.8}
&
Provides customizable bottom sheet components that can be expanded or collapsed to display additional content without leaving the current screen. It is used mainly in the map component to render collections.
\\ \hline

\packagename{@react-native-community/geolocation}~\cite{site:rn-geolocation}
\newline
\textit{Version:} \texttt{\^{}3.4.0}
&
Provides access to the device's geolocation services, enabling the application to retrieve the user's current position.
\\ \hline

\packagename{@react-native-community/slider}~\cite{site:rn-slider}
\newline
\textit{Version:} \texttt{\^{}5.2.0}
&
Offers a slider component that allows users to select a value within a specified range, used to choose the distance range for post visibility.
\\ \hline

\packagename{@shopify/flash-list}~\cite{site:rn-flashlist}
\newline
\textit{Version:} \texttt{\^{}2.3.1}
&
Offers a high-performance list component optimized for rendering large collections of data with improved memory usage and scrolling performance.
\\ \hline


\packagename{expo}~\cite{site:rn-expo}
\newline
\textit{Version:} \texttt{\~{}54.0.33}
&
provides the development framework and runtime environment for building, testing and deploying React Native applications, together with a wide range of native APIs.
\\ \hline


\packagename{expo-router}~\cite{site:rn-exporouter}
\newline
\textit{Version:} \texttt{\~{}6.0.23}
&
Implements a file-based routing system that simplifies navigation between the different screens of the application.
\\ \hline

\packagename{expo-secure-store}~\cite{rn-expostore}
\newline
\textit{Version:} \texttt{\~{}15.0.8}
&
Securely stores sensitive information, such as user credentials, by leveraging the secure storage facilities provided by the underlying operating system.
\\ \hline

\packagename{react-native-gesture-handler}~\cite{site:rn-gesturehandler}
\newline
\textit{Version:} \texttt{\~{}2.28.0}
&
Enables advanced gesture recognition, including swipes, taps, and drag interactions, providing a more responsive and native user experience.
\\ \hline

\packagename{react-native-gifted-charts}~\cite{site:rn-charts}
\newline
\textit{Version:} \texttt{\^{}1.4.76}
&
Provides customizable chart components for visualizing statistical information about usage on the user profile.
\\ \hline

\packagename{react-native-maps}~\cite{site:rn-maps}
\newline
\textit{Version:} \texttt{1.20.1}
&
Integrates interactive maps into the application, allowing the visualization of geographic information and user locations.
\\ \hline

\packagename{react-native-reanimated-carousel}~\cite{site:rn-carousel}
\newline
\textit{Version:} \texttt{\^{}4.0.3}
&
Implements an animated carousel component for displaying files attached to a post through horizontal scrolling.
\\ \hline

\packagename{react-native-share}~\cite{site:rn-share}
\newline
\textit{Version:} \texttt{\^{}12.3.1}
&
Uses the native sharing facilities of the operating system to share or store media attached to a post, such as images or documents.
\\ \hline

\packagename{react-native-sortables}~\cite{site:rn-shasortables}
\newline
\textit{Version:} \texttt{\^{}1.9.4}

Provides sortable list components that allow users to reorder items through drag-and-drop interactions. It is used to arrange posts whenever sequential features are active in a collection.
\\ \hline

\packagename{react-native-vision-camera}~\cite{site:rn-camera}

\textit{Version:} \texttt{\^{}4.7.3}
